## 🔨 构建配置

### Makefile 配置

```makefile
# IPTV M3U Aggregator - Makefile

.PHONY: help build build-windows build-debian clean test

# 变量定义
BINARY_NAME=iptv-aggregator
VERSION=$(shell git describe --tags --always --dirty)
BUILD_TIME=$(shell date -u '+%Y-%m-%d %H:%M:%S')
GIT_COMMIT=$(shell git rev-parse --short HEAD)

# Windows 构建参数
WINDOWS_BINARY=$(BINARY_NAME).exe
WINDOWS_OUTPUT=build/windows/$(WINDOWS_BINARY)

# Debian 构建参数
DEBIAN_BINARY=$(BINARY_NAME)
DEBIAN_OUTPUT=build/debian/$(DEBIAN_BINARY)

# 编译标志
LDFLAGS=-ldflags "-X main.Version=$(VERSION) -X main.BuildTime=$(BUILD_TIME) -X main.GitCommit=$(GIT_COMMIT)"

help:
	@echo "IPTV M3U Aggregator - Build Commands"
	@echo ""
	@echo "Usage:"
	@echo "  make build              - Build for current platform"
	@echo "  make build-windows      - Build for Windows x64"
	@echo "  make build-debian       - Build for Debian/Linux"
	@echo "  make build-all          - Build for all platforms"
	@echo "  make clean              - Clean build artifacts"
	@echo "  make test               - Run tests"
	@echo "  make test-race          - Run tests with race detector"
	@echo "  make coverage           - Generate coverage report"
	@echo ""

# 当前平台构建
build:
	@echo "Building for current platform..."
	go build $(LDFLAGS) -o $(BINARY_NAME) .

# Windows x64 构建
build-windows:
	@echo "Building for Windows x64..."
	@mkdir -p build/windows
	CGO_ENABLED=0 GOOS=windows GOARCH=amd64 go build $(LDFLAGS) -o $(WINDOWS_OUTPUT) .
	@echo "✓ Windows binary: $(WINDOWS_OUTPUT)"

# Debian/Linux x64 构建
build-debian:
	@echo "Building for Debian/Linux x64..."
	@mkdir -p build/debian
	CGO_ENABLED=0 GOOS=linux GOARCH=amd64 go build $(LDFLAGS) -o $(DEBIAN_OUTPUT) .
	@chmod +x $(DEBIAN_OUTPUT)
	@echo "✓ Debian binary: $(DEBIAN_OUTPUT)"

# 构建所有平台
build-all: build-windows build-debian
	@echo "✓ All platforms built successfully"

# 清理构建文件
clean:
	@echo "Cleaning build artifacts..."
	@rm -rf build/
	@rm -f $(BINARY_NAME) $(BINARY_NAME).exe
	@echo "✓ Clean complete"

# 运行测试
test:
	@echo "Running tests..."
	go test -v ./...

# 运行竞态检测
test-race:
	@echo "Running tests with race detector..."
	go test -race -v ./...

# 生成覆盖率报告
coverage:
	@echo "Generating coverage report..."
	go test -coverprofile=coverage.out ./...
	go tool cover -html=coverage.out -o coverage.html
	@echo "✓ Coverage report: coverage.html"

# 代码格式化
fmt:
	@echo "Formatting code..."
	go fmt ./...
	@echo "✓ Format complete"

# 静态分析
lint:
	@echo "Running linter..."
	go vet ./...
	@echo "✓ Lint complete"

# 依赖更新
deps:
	@echo "Updating dependencies..."
	go get -u ./...
	go mod tidy
	@echo "✓ Dependencies updated"

# 完整构建流程
all: clean test build-all
	@echo "✓ Complete build successful"
```

## 🪟 Windows x64 构建

### 前置条件

```bash
# 安装 Go 1.21+
# 下载地址: https://golang.org/dl/

# 验证安装

```

### 构建步骤

#### 方法 1：使用 Makefile（推荐）

```bash
# 在项目根目录执行
make build-windows

# 输出
# Building for Windows x64...
# ✓ Windows binary: build/windows/iptv-aggregator.exe
```


## 🐧 Debian/Linux x64 构建

### 前置条件

```bash
# 安装 Go 1.21+
wget https://go.dev/dl/go1.21.0.linux-amd64.tar.gz
sudo tar -C /usr/local -xzf go1.21.0.linux-amd64.tar.gz
export PATH=$PATH:/usr/local/go/bin

# 验证安装
go version
```

### 构建步骤

#### 方法 1：使用 Makefile（推荐）

```bash
# 在项目根目录执行
make build-debian

# 输出
# Building for Debian/Linux x64...
# ✓ Debian binary: build/debian/iptv-aggregator
```

#### 方法 2：手动构建

```bash
# 创建输出目录
mkdir -p build/debian

# 构建 Debian 二进制
CGO_ENABLED=0 GOOS=linux GOARCH=amd64 go build -o build/debian/iptv-aggregator .

# 添加执行权限
chmod +x build/debian/iptv-aggregator
```

### 构建输出

```
build/debian/
└── iptv-aggregator                 # 可执行文件（~15-20 MB）
```

### Debian 部署

#### 1. 准备目录结构

```bash
# 创建应用目录
sudo mkdir -p /opt/iptv-aggregator
sudo mkdir -p /opt/iptv-aggregator/data
sudo mkdir -p /opt/iptv-aggregator/data/exports

# 复制二进制文件
sudo cp build/debian/iptv-aggregator /opt/iptv-aggregator/
sudo chmod +x /opt/iptv-aggregator/iptv-aggregator

# 创建配置文件
sudo cp config.json /opt/iptv-aggregator/
```

#### 2. 创建 Systemd 服务

```bash
# 创建服务文件
sudo tee /etc/systemd/system/iptv-aggregator.service > /dev/null <<EOF
[Unit]
Description=IPTV M3U Aggregator
After=network.target

[Service]
Type=simple
User=iptv
WorkingDirectory=/opt/iptv-aggregator
ExecStart=/opt/iptv-aggregator/iptv-aggregator
Restart=on-failure
RestartSec=10

# 日志
StandardOutput=journal
StandardError=journal

# 资源限制
MemoryLimit=200M
CPUQuota=50%

[Install]
WantedBy=multi-user.target
EOF

# 创建用户
sudo useradd -r -s /bin/false iptv

# 设置权限
sudo chown -R iptv:iptv /opt/iptv-aggregator

# 重新加载 systemd
sudo systemctl daemon-reload

# 启用服务
sudo systemctl enable iptv-aggregator

# 启动服务
sudo systemctl start iptv-aggregator

# 查看服务状态
sudo systemctl status iptv-aggregator

# 查看日志
sudo journalctl -u iptv-aggregator -f
```

#### 3. 创建 Nginx 反向代理（可选）

```nginx
# /etc/nginx/sites-available/iptv-aggregator
upstream iptv_backend {
    server 127.0.0.1:8080;
}

server {
    listen 80;
    server_name iptv.example.com;

    # 重定向到 HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name iptv.example.com;

    # SSL 证书
    ssl_certificate /etc/letsencrypt/live/iptv.example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/iptv.example.com/privkey.pem;

    # 代理设置
    location / {
        proxy_pass http://iptv_backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # WebSocket 支持
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    # 静态文件缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        proxy_pass http://iptv_backend;
        proxy_cache_valid 200 30d;
        expires 30d;
    }
}
```

启用站点：
```bash
sudo ln -s /etc/nginx/sites-available/iptv-aggregator /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl restart nginx
```

#### 4. 创建 Docker 容器（可选）

```dockerfile
# Dockerfile
FROM golang:1.21-alpine AS builder

WORKDIR /app
COPY . .

RUN go mod download
RUN CGO_ENABLED=0 GOOS=linux GOARCH=amd64 go build -ldflags="-s -w" -o iptv-aggregator .

# 运行阶段
FROM alpine:latest

RUN apk --no-cache add ca-certificates tzdata

WORKDIR /app

# 创建非 root 用户
RUN addgroup -g 1000 appuser && \
    adduser -D -u 1000 -G appuser appuser

COPY --from=builder /app/iptv-aggregator .
RUN mkdir -p data && chown -R appuser:appuser /app

USER appuser

EXPOSE 8080

CMD ["./iptv-aggregator"]
```

构建和运行：
```bash
# 构建镜像
docker build -t iptv-aggregator:latest .

# 运行容器
docker run -d \
  --name iptv-aggregator \
  -p 8080:8080 \
  -v /opt/iptv-aggregator/data:/app/data \
  -v /opt/iptv-aggregator/config.json:/app/config.json:ro \
  iptv-aggregator:latest

# 查看日志
docker logs -f iptv-aggregator

# 停止容器
docker stop iptv-aggregator

# 删除容器
docker rm iptv-aggregator
```

## 📦 构建脚本

### build.sh（Linux/Mac）

```bash
#!/bin/bash

set -e

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 函数定义
print_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

print_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

print_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $1"
}

# 检查 Go 安装
if ! command -v go &> /dev/null; then
    print_error "Go is not installed"
    exit 1
fi

print_info "Go version: $(go version)"

# 清理旧的构建
print_info "Cleaning old builds..."
rm -rf build/

# 构建 Windows
print_info "Building for Windows x64..."
mkdir -p build/windows
CGO_ENABLED=0 GOOS=windows GOARCH=amd64 go build -o build/windows/iptv-aggregator.exe .
print_info "✓ Windows binary: build/windows/iptv-aggregator.exe"

# 构建 Debian
print_info "Building for Debian/Linux x64..."
mkdir -p build/debian
CGO_ENABLED=0 GOOS=linux GOARCH=amd64 go build -o build/debian/iptv-aggregator .
chmod +x build/debian/iptv-aggregator
print_info "✓ Debian binary: build/debian/iptv-aggregator"

print_info "✓ All builds completed successfully"
```

### build.bat（Windows）

```batch
@echo off
setlocal enabledelayedexpansion

REM 颜色定义
set "GREEN=[92m"
set "RED=[91m"
set "YELLOW=[93m"
set "NC=[0m"

REM 检查 Go 安装
go version >nul 2>&1
if errorlevel 1 (
    echo %RED%[ERROR]%NC% Go is not installed
    exit /b 1
)

echo %GREEN%[INFO]%NC% Go version:
go version

REM 清理旧的构建
echo %GREEN%[INFO]%NC% Cleaning old builds...
if exist build rmdir /s /q build

REM 构建 Windows
echo %GREEN%[INFO]%NC% Building for Windows x64...
if not exist build\windows mkdir build\windows
set CGO_ENABLED=0
set GOOS=windows
set GOARCH=amd64
go build -o build\windows\iptv-aggregator.exe .
echo %GREEN%[INFO]%NC% Windows binary: build\windows\iptv-aggregator.exe

REM 构建 Debian（需要在 Linux/Mac 上运行）
echo %YELLOW%[WARNING]%NC% Debian build requires Linux/Mac environment
echo %YELLOW%[WARNING]%NC% Use build.sh on Linux/Mac to build Debian binary

echo %GREEN%[INFO]%NC% Build completed successfully
```

## 📝 故障排查

### Windows 常见问题

**Q: 无法运行 .exe 文件**
A: 检查是否有 Windows Defender 或防火墙阻止

**Q: 端口被占用**
A: 修改 config.json 中的 port 配置

**Q: 找不到 config.json**
A: 确保 config.json 与 .exe 在同一目录

### Debian 常见问题

**Q: Permission denied**
A: 运行 `chmod +x iptv-aggregator`

**Q: 服务无法启动**
A: 运行 `sudo journalctl -u iptv-aggregator -n 50` 查看错误日志

**Q: 端口被占用**
A: 运行 `sudo lsof -i :8080` 查看占用进程
